const state = { layout:"pairs", students:[], constraints:[], candidates:[], selectedCandidate:null, currentArrangement:[], lockedStudents:new Set(), activeTab:"students" };

const demoNames = ["Ava","Ben","Carla","Diego","Elena","Finn","Grace","Hassan","Ivy","Jake","Kayla","Leo","Maya","Noah","Olivia","Priya","Quinn","Riley","Sofia","Theo","Uma","Victor","Willow","Zane"];
function makeDemoStudents(){
  return demoNames.map((name,i)=>({ id:crypto.randomUUID(), name,
    visibility:["medium","high","medium","low","medium","high"][i%6],
    behavior:["low","medium","high","medium","low","medium"][i%6] }));
}

const $ = s => document.querySelector(s);
const esc = v => String(v).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
const norm = n => String(n||"").trim();
const valueScore = v => v==="low"?1:v==="medium"?2:3;

function groupSizes(n){
  if(n<=0) return [];
  const groups = Math.ceil(n/5);
  const sizes = Array(groups).fill(5);
  let extra = groups*5 - n;
  let idx = groups-1;
  while(extra>0){ if(sizes[idx]>4){sizes[idx]--; extra--;} else { idx--; if(idx<0) break; } }
  while(sizes.some(s=>s<4) && groups>1){ sizes[sizes.indexOf(5)]--; sizes[sizes.lastIndexOf(4)]++; }
  return sizes;
}

function renderStats(){
  $("#studentTabCount").textContent = state.students.length;
  $("#ruleTabCount").textContent = state.constraints.length;
}

function renderStudents(){
  const box = $("#studentList"); box.innerHTML = "";
  state.students.forEach(st=>{
    const row = document.createElement("div");
    row.className = "student-row" + (state.lockedStudents.has(st.id) ? " locked":"");
    row.innerHTML = `
      <input class="student-name" value="${esc(st.name)}" />
      <select class="mini-select vis">
        <option value="low" ${st.visibility==="low"?"selected":""}>Vis L</option>
        <option value="medium" ${st.visibility==="medium"?"selected":""}>Vis M</option>
        <option value="high" ${st.visibility==="high"?"selected":""}>Vis H</option>
      </select>
      <select class="mini-select beh">
        <option value="low" ${st.behavior==="low"?"selected":""}>Beh L</option>
        <option value="medium" ${st.behavior==="medium"?"selected":""}>Beh M</option>
        <option value="high" ${st.behavior==="high"?"selected":""}>Beh H</option>
      </select>
      <button class="icon-sm lock ${state.lockedStudents.has(st.id)?"on":""}" title="Lock seat">${state.lockedStudents.has(st.id)?"●":"○"}</button>

    row.querySelector(".student-name").addEventListener("input", e=>{ st.name=e.target.value; renderConstraints(); renderClassTitle(); });
    row.querySelector(".vis").addEventListener("change", e=> st.visibility=e.target.value);
    row.querySelector(".beh").addEventListener("change", e=> st.behavior=e.target.value);
    row.querySelector(".lock").addEventListener("click", ()=>{
      state.lockedStudents.has(st.id) ? state.lockedStudents.delete(st.id) : state.lockedStudents.add(st.id);
      renderStudents(); renderClassroom(state.currentArrangement);
    });
    row.querySelector(".del").addEventListener("click", ()=>{
      state.lockedStudents.delete(st.id);
      state.constraints = state.constraints.filter(c=>c[0]!==st.id && c[1]!==st.id);
      state.students = state.students.filter(s=>s.id!==st.id);
      renderAll();
    });
    box.appendChild(row);
  });
  renderStats();
}

function fillOptions(sel){
  sel.innerHTML = state.students.map(s=>`<option value="${esc(s.id)}">${esc(norm(s.name)||"Unnamed")}</option>`).join("");
}

function renderConstraints(){
  const box = $("#constraintList"); box.innerHTML = "";
  if(!state.constraints.length){

  }
  state.constraints.forEach((c,i)=>{

    sel.addEventListener("change", e=> state.constraints[i][2]=e.target.value);

    wrap.append(row, srow);
    box.appendChild(wrap);
  });
}

function addStudent(){
  state.students.push({ id:crypto.randomUUID(), name:`Student ${state.students.length+1}`, visibility:"medium", behavior:"medium" });
  renderStudents(); renderConstraints();

function addConstraint(){
  if(state.students.length<2) return;
  state.constraints.push([state.students[0].id, state.students[1].id, "hard"]);
  renderConstraints(); renderStats();
}

function buildUnits(){
  const units = [];
  if(state.layout==="pairs"){ for(let i=0;i<state.students.length;i+=2) units.push(state.students.slice(i,i+2)); }
  else { const sizes=groupSizes(state.students.length); let c=0; sizes.forEach(sz=>{ units.push(state.students.slice(c,c+sz)); c+=sz; }); }
  return units;
}
function shuffle(arr, rng=Math.random){ const a=[...arr]; for(let i=a.length-1;i>0;i--){ const j=Math.floor(rng()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; }
function seeded(n){ const x=Math.sin(n*999.91)*10000; return x-Math.floor(x); }

function calcSeatQuality(st, unitIndex, total){
  const pos = total<=1 ? 0 : unitIndex/(total-1);
  const pref = (4-valueScore(st.visibility))/3;
  return 1 - Math.abs(pos-pref)*0.55;
}
function arrangementPenalty(arr){
  const map=new Map(); arr.forEach((u,ui)=>u.forEach(s=>map.set(s.id,ui)));
  let hard=0, soft=0;
  for(const [a,b,strength] of state.constraints){
    if(!map.has(a)||!map.has(b)) continue;
    if(map.get(a)===map.get(b)){ strength==="hard" ? hard++ : soft++; }
  }
  return {hard, soft};
}
function scoreArrangement(arr){
  const total = arr.length; let vis=0, beh=0, locked=0;
  arr.forEach((u,ui)=>u.forEach(s=>{
    vis += calcSeatQuality(s,ui,total)*100;
    beh += 100 - (valueScore(s.behavior)-1)*8;
    if(state.lockedStudents.has(s.id)) locked++;
  }));
  const n = Math.max(state.students.length,1);
  vis/=n; beh/=n;
  const p = arrangementPenalty(arr);
  const score = Math.max(0, Math.round(vis*0.5 + beh*0.3 + (p.hard===0?15:Math.max(0,15-p.hard*7)) + Math.max(0,5-p.soft*2)));
  return { score, visibility:Math.round(vis), behavior:Math.round(beh), hard:p.hard, soft:p.soft, locked };
}
function wouldConflict(student, unit){
  return state.constraints.some(([a,b,strength])=>{
    if(strength!=="hard") return false;
    const ids = new Set(unit.map(s=>s.id));
    return (a===student.id && ids.has(b)) || (b===student.id && ids.has(a));
  });
}
function createCandidate(seed){
  let working = shuffle(state.students, ()=>seeded(seed));
  working.sort((a,b)=> valueScore(b.visibility)-valueScore(a.visibility));
  const unitCount = state.layout==="pairs" ? Math.ceil(working.length/2) : groupSizes(working.length).length;
  const units = Array.from({length:unitCount}, ()=>[]);
  const capacities = state.layout==="pairs" ? units.map(()=>2) : groupSizes(state.students.length);
  const lockedStudents = state.students.filter(s=>state.lockedStudents.has(s.id));
  const lockedPlacement = new Map();
  if(state.currentArrangement.length){
    state.currentArrangement.forEach((u,ui)=>u.forEach((s,si)=>{ if(state.lockedStudents.has(s.id)) lockedPlacement.set(s.id,[ui,si]); }));
  }
  for(const s of lockedStudents){
    const p = lockedPlacement.get(s.id);
    if(p){ const [ui,si]=p; if(!units[ui]) units[ui]=[]; units[ui][si]=s; }
  }
  const unplaced = working.filter(s=>!lockedStudents.some(l=>l.id===s.id));
  for(const s of shuffle(unplaced, ()=>seeded(seed + s.id.length))){
    const choices = units.map((u,ui)=>({ui,unit:u,cap:capacities[ui]||2})).filter(x=>x.unit.filter(Boolean).length<x.cap);
    choices.sort((a,b)=>{
      const ah=wouldConflict(s,a.unit)?1:0, bh=wouldConflict(s,b.unit)?1:0;
      if(ah!==bh) return ah-bh;
      const ab=a.unit.filter(Boolean).filter(x=>x.behavior==="high").length;
      const bb=b.unit.filter(Boolean).filter(x=>x.behavior==="high").length;
      if(ab!==bb) return ab-bb;
      return a.unit.filter(Boolean).length - b.unit.filter(Boolean).length;
    });
    const choice = choices[0]; if(!choice) continue;
    const slot = choice.unit.findIndex(x=>!x);
    if(slot>=0) choice.unit[slot]=s; else choice.unit.push(s);
  }
  return units.map(u=>u.filter(Boolean));
}

function generateCandidates(){
  if(!state.students.length) return;
  const cands = [];
  for(let i=0;i<8;i++){
    const arr = createCandidate(41+i*17);
    cands.push({ id:crypto.randomUUID(), name:`Option ${String.fromCharCode(65+i)}`, arrangement:arr, ...scoreArrangement(arr) });
  }
  cands.sort((a,b)=>b.score-a.score);
  state.candidates = cands.slice(0,5);
  state.selectedCandidate = state.candidates[0]?.id || null;
  state.currentArrangement = state.candidates[0]?.arrangement || [];
  renderCandidates(); renderClassroom(state.currentArrangement); renderInsight(state.candidates[0]);
}

function renderCandidates(){
  const list = $("#candidateList");
  $("#candidateSub").textContent = state.candidates.length ? `${state.candidates.length} options generated` : "Generate a few options, then compare tradeoffs.";
  if(!state.candidates.length){ list.innerHTML=""; return; }
  list.innerHTML = state.candidates.map(c=>`
    <div class="cand ${c.id===state.selectedCandidate?"sel":""}" data-id="${c.id}">
      <div class="cand-score">${c.score}</div>
      <div><div class="cand-title">${esc(c.name)}</div>
      <div class="cand-detail">Visibility ${c.visibility}% · Behavior ${c.behavior}% · ${c.hard} hard conflicts</div></div>
      <div class="cand-tag">${c.hard===0?"Clean":"Review"}</div>
    </div>`).join("");
  list.querySelectorAll(".cand").forEach(el=>{
    el.addEventListener("click", ()=>{
      state.selectedCandidate = el.dataset.id;
      const c = state.candidates.find(x=>x.id===state.selectedCandidate);
      state.currentArrangement = c.arrangement;
      renderCandidates(); renderClassroom(c.arrangement); renderInsight(c);
    });
  });
}

function renderInsight(c){
  const box = $("#insightBox");
  if(!c){ box.innerHTML = `<div class="insight-score"><span>—</span><small>/100</small></div><p class="panel-sub">Select a candidate to see what the planner prioritized.</p>`; return; }
  const notes = [
    `${c.visibility}% visibility fit across the class.`,
    `${c.behavior}% behavior-balancing score.`,
    c.hard===0 ? "All hard separation rules are satisfied." : `${c.hard} hard separation conflict${c.hard===1?"":"s"} remain.`
  ];
  if(c.soft>0) notes.push(`${c.soft} soft preference conflict${c.soft===1?"":"s"} remain.`);
  if(c.locked>0) notes.push(`${c.locked} locked student${c.locked===1?"":"s"} were kept in place.`);
  box.innerHTML = `<div class="insight-score"><span>${c.score}</span><small>/100</small></div>
    <p class="panel-sub">A starting recommendation, not a final decision.</p>
    <ul class="insight-list">${notes.map(n=>`<li>${esc(n)}</li>`).join("")}</ul>`;
}

function renderClassroom(arr){
  const el = $("#classroom");
  el.innerHTML = "";
  if(!arr.length){ el.innerHTML = `<div class="empty-room">Add students and press Generate to see the room.</div>`; return; }
  arr.forEach((unit,ui)=>{
    const u = document.createElement("div"); u.className="unit";
    u.innerHTML = `<div class="unit-label">${state.layout==="pairs"?"DESK":"GROUP"} ${ui+1}</div>`;
    const seats = document.createElement("div"); seats.className="seats";
    unit.forEach(st=>{
      const seat = document.createElement("div"); seat.className="seat";
      seat.innerHTML = `<span class="seat-name">${esc(norm(st.name)||"Unnamed")}</span>
        <span>${state.lockedStudents.has(st.id)?'<span class="badge lock">LOCK</span>':''}
        <span class="badge">${st.visibility[0].toUpperCase()}</span></span>`;
      seats.appendChild(seat);
    });
    u.appendChild(seats); el.appendChild(u);
  });
}
